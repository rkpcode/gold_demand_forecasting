# Step 1: Generate date range (1 year dummy data)
dates = pd.date_range(start="2023-01-01", end="2023-12-31")

# Step 2: Generate dummy sales (grams per day)
np.random.seed(42)
base_sales = np.random.randint(50, 150, size=len(dates))  # random base sales
seasonal_boost = [50 if d.month in [10, 11] else 0 for d in dates]  # festive season boost
sales_qty = base_sales + seasonal_boost

# Step 3: Stock level (assume starting stock 5000g, reduce daily sales, restock occasionally)
stock = []
current_stock = 5000
for s in sales_qty:
    current_stock -= s
    if current_stock < 1000:  # restock condition
        current_stock += 3000
    stock.append(current_stock)

# Step 4: Gold price simulation (random walk around ₹5500/g)
gold_price = 5500 + np.cumsum(np.random.normal(0, 10, len(dates)))

# Step 5: Festival flag (assume some fixed dates)
festivals = ["2023-01-14", "2023-03-08", "2023-10-22", "2023-11-12"]
festival_flag = [1 if str(d.date()) in festivals else 0 for d in dates]

# Step 6: Weekday (0=Mon, 6=Sun)
weekday = [d.weekday() for d in dates]

# Create DataFrame
df = pd.DataFrame({
    "Date": dates,
    "SalesQty": sales_qty,
    "StockLevel": stock,
    "GoldPrice": gold_price,
    "FestivalFlag": festival_flag,
    "Weekday": weekday
})

import pandas as pd
import numpy as np
from prophet import Prophet
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_absolute_error, mean_squared_error
import matplotlib.pyplot as plt

# ---------- Step 1: Prepare Data ----------
sales_series = df[["Date", "SalesQty"]].set_index("Date")
train = sales_series[:-60]
test = sales_series[-60:]

# ---------- Step 2: Prophet ----------
df_prophet = train.reset_index().rename(columns={"Date":"ds","SalesQty":"y"})

prophet_model = Prophet(yearly_seasonality=True, weekly_seasonality=True)
prophet_model.fit(df_prophet)

future = prophet_model.make_future_dataframe(periods=60)
forecast = prophet_model.predict(future)

prophet_pred = forecast.tail(60)["yhat"].values

prophet_mae = mean_absolute_error(test, prophet_pred)
prophet_rmse = np.sqrt(mean_squared_error(test, prophet_pred))

# ---------- Step 3: ARIMA ----------
arima_model = ARIMA(train, order=(5,1,0))
arima_fit = arima_model.fit()
arima_pred = arima_fit.forecast(steps=60)

arima_mae = mean_absolute_error(test, arima_pred)
arima_rmse = np.sqrt(mean_squared_error(test, arima_pred))

# ---------- Step 4: Compare ----------
print("Prophet -> MAE:", prophet_mae, "RMSE:", prophet_rmse)
print("ARIMA   -> MAE:", arima_mae, "RMSE:", arima_rmse)

if prophet_rmse < arima_rmse:
    print("✅ Best Model: Prophet")
    best_pred = prophet_pred
else:
    print("✅ Best Model: ARIMA")
    best_pred = arima_pred

# ---------- Plot ----------
plt.figure(figsize=(12,6))
plt.plot(train.index, train.values, label="Train")
plt.plot(test.index, test.values, label="Test")
plt.plot(test.index, best_pred, label="Best Model Prediction", color="red")

def decision_engine(df, forecasted_demand, today_date, safety_stock=200):
    # Extract today’s values
    today_data = df[df["Date"] == today_date].iloc[0]
    current_stock = today_data["StockLevel"]
    today_price = today_data["GoldPrice"]

    # Forecasted demand (next 7 days sum)
    demand_next7 = forecasted_demand[:7].sum()

    # Gold price moving avg (last 7 days)
    price_7d_avg = df[df["Date"] < today_date].tail(7)["GoldPrice"].mean()

    # Festival adjustment
    if today_data["FestivalFlag"] == 1:
        demand_next7 *= 1.2  # +20%

    # Required purchase calculation
    required_purchase = demand_next7 - current_stock + safety_stock

    # Business rules
    if required_purchase <= 0:
        recommendation = "❌ No Purchase Needed"
        buy_qty = 0
    else:
        if today_price < price_7d_avg:
            recommendation = "✅ Strong Buy"
            buy_qty = required_purchase
        else:
            recommendation = "⚠️ Buy Minimum Buffer Only"
            buy_qty = safety_stock  # just maintain buffer

    return {
        "Date": today_date,
        "Recommendation": recommendation,
        "BuyQty": round(buy_qty, 2),
        "Reason": f"Forecasted Demand={round(demand_next7,2)}, Stock={current_stock}, Price={round(today_price,2)}, 7dAvg={round(price_7d_avg,2)}"
    }

# Example run
today = "2023-11-01"
result = decision_engine(df, best_pred, today_date=today)
print(result)
